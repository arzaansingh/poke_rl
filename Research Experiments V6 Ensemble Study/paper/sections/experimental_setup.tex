\section{Experimental Setup}
\label{sec:experiments}

\subsection{Environment}
\label{sec:env}

\guideblock{One short paragraph. State the stack: \texttt{poke-env}
\citep{sahovic2020pokeenv} on top of Pokemon Showdown, \texttt{gen4ou}
format, fixed 20-Pokemon pool, \texttt{IndexedTeambuilder} deterministic
team sampling (so members and opponents see identical matchups at the
same battle index). Pool composition and the deterministic team-sampling
recipe live in Appendix~\ref{app:reproducibility}; no need to repeat them
here.}

\subsection{Evaluation conditions}
\label{sec:eval_conditions}

\guideblock{Keep this list as-is in the final draft (numerics are locked).
Voting strategies at $K{=}10$ (soft / hard / confidence); $K$-saturation
sweep at soft voting ($K\in\{1,3,5,10\}$); head-to-head ensemble
vs.\ \texttt{baseline\_5m}; \texttt{baseline\_5m} solo vs.\
\texttt{SimpleHeuristicsPlayer}.}

\begin{itemize}
  \item \textbf{Voting strategies at $K=10$}: SOFT, HARD, CONFIDENCE, each vs.\ \texttt{SimpleHeuristicsPlayer}.
  \item \textbf{$K$-saturation curve at SOFT}: $K \in \{1, 3, 5, 10\}$ vs.\ \texttt{SimpleHeuristicsPlayer}.
  \item \textbf{Head-to-head}: $K{=}10$ SOFT ensemble vs.\ \texttt{baseline\_5m} (compute-matched single agent).
  \item \textbf{Baseline solo}: \texttt{baseline\_5m} solo vs.\ \texttt{SimpleHeuristicsPlayer}.
\end{itemize}

\subsection{Evaluation protocol}
\label{sec:eval_protocol}

\guideblock{Keep the numerics. $N = 100{,}000$ battles per condition,
Wilson 95\% intervals \citep{wilson1927}, normal-approximation
difference-of-proportions test for between-condition comparisons. Brief
prose pass to integrate this into the paragraph after the meeting.}

Each condition is evaluated over $N = 100{,}000$ battles. Win rates are
reported with Wilson 95\% confidence intervals \citep{wilson1927}.
Differences in proportions use the normal approximation to the
difference of independent binomials. All hardware specs and elapsed
wall-clock times are tabulated in \cref{app:reproducibility}.

\subsection{Baseline opponents}
\label{sec:baselines}

\guideblock{One paragraph for each baseline. (i)
\texttt{SimpleHeuristicsPlayer}: \texttt{poke-env}'s reference heuristic
\citep{sahovic2020pokeenv}; standard opponent in
\citet{wang2024mit}, \citet{grigsby2025metamon}, \citet{vgcbench2025}.
(ii) \texttt{baseline\_5m}: a single \texttt{HierSmartPlayer} trained for
$5{\times}10^6$ battles ($5{\times}$ each ensemble member's budget)
under identical hyperparameters, hardware, and pool; the
\emph{compute-matched} single-agent control. Make this distinction
explicit --- without it the $+12.6$ pp lift in
Section~\ref{sec:r_headline} could be attributed to extra wall-clock
compute rather than to ensembling.}
